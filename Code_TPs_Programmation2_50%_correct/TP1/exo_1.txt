int* Allouer(int taille) {
	int* T;
	T=(int*)malloc(taille*sizeof(int));
	return T; 
	}

/* ---------------Exercice1-Question2---------------------*/
void saisie(int* T, int taille) {
	int i;
	for(i=0;i<taille;i++) {
		printf("Entrer l element du tableau numero %d :\t",i);
		scanf("%d",T+i); 
		} 
	}
/* ---------------Exercice1-Question3---------------------*/
void affiche(int* T, int taille) {
	int i;
	printf("les elements du tableau sont:\n");
	for(i=0;i<taille;i++){
		printf("%d\t",*(T+i)); } 
	}
/* ---------------Exercice1-Question4---------------------*/
int * Reallouer(int* T, int taille) {
	T=(int *)realloc(T,taille*sizeof(int));
	return T; 
	}
	
/* ---------------Exercice1-Question5---------------------*/
void insereValeurs(int * T, int *taille, int ind, int N) {
	int i,j;
	if((ind>*taille)||(ind<0)) printf("Insertion impossible!!!! \n\n");
	else {
		for(j=0;j<=N;j++) {
			if (j%2!=0) {
				(*taille)++;
				T=Reallouer(T,(*taille));
				for(i=(*taille)-1;i>=(ind-1);i--) {
					*(T+(i+1))= *(T+i);
				}
				*(T+ind)=j;
				ind=ind+1;
				}		
			}
		} 
	}
	
/* ---------------Exercice1-Question6---------------------*/
void tribulle(int * T, int N) {
	int trie, i, tmp;
	do{
		trie = 1;
		for(i=1;i<N;i++){
			if( *(T+i-1) > *(T+i)){
				tmp = *(T+i);
				*(T+i) = *(T+i-1);
				*(T+i-1) = tmp;
				trie = 0; 
				} 
			}
	} while (trie==0); 
}

/* ---------------Exercice1-Question7---------------------*/
void supprimeValeur(int * T, int *taille){
	int i,j; int k;
	T=Reallouer(T, (*taille));
	// Supprimer les doublons
	int fin=0;
	while (fin!=1) {
		fin=1;
		for (i=0;i<(*taille)-1;i++) {
			for(j=i+1;j<(*taille);) {
				if (*(T+i)==*(T+j)) {
					for (k = j; k < (*taille); k++) {
						*(T+k) = *(T+k+1); 
					}
				fin = 0;
				(*taille)--; 
				}
				else{ 
				j++;
				}
			} 
		} 
	}
	T=Reallouer(T, (*taille)); 
	}

void compareTableaux (int *T1, int n1, int *T2, int n2) {
	int i,j, C=0;
	if (n1==n2)
	{
		for (i=0;i<n1;i++){
			if (*(T1+i)==*(T2+i)) C++; 
		}
		if(C==n1) printf("\n ************** Identiques\n");
		else if (C==0) printf("\n ************** Ne coincident pas\n");
		else printf("\n ************** Coincident\n"); 
	}
	else {
		for (i=0;i<n1;i++){
			for (j=0;j<n2;j++){
				if (*(T1+i)==*(T2+j)) C++; 
			}
		}
		if(C==0)
		printf("\n ************** Ne coincident pas\n");
		else printf("\n ************** Coincident\n"); } 
}
