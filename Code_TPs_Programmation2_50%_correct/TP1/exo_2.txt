/* ---------------Exercice2-Question1---------------------*/
int ** allouer( int L , int C) {
	int **T;
	int i;
	T=(int **)malloc(L*sizeof(int));
	for(i=0;i<L;i++) {
		T[i]=(int*)(malloc(C*sizeof(int))); 
		}
		return T; 
	}

/* ---------------Exercice2-Question2---------------------*/
void saisie(int ** T, int L, int C) {
	int i,j;
	for(i=0;i<L;i++) {
		for(j=0;j<C;j++) {
			printf("T[%d][%d]= ",i, j);
			scanf("%d",(*(T+i)+j)); 
			}
			printf("\n");
		} 
	}

void Affiche(int ** T, int L, int C) {
	int i,j;
	printf("la forme matricielle de M est : \n");
	for(i=0;i<L;i++) {
		for(j=0;j<C;j++) {
		printf("%d\t",*(*(T+i)+j));
		}
		printf("\n"); 
		} 
	}

int CarreMagique(int ** T, int L, int C) {
	int OK = 1;
	int i;
	int j;
	int somme = 0;
	int sommeSuiv = 0;
	//initialisation de la somme pour la première ligne
	for (j=0; j< L ; j++)	somme += T[0][j];
	//comparaison avec les autres lignes
	i= 1;
	while (OK && i < L) {
		sommeSuiv = 0;
		for (j=0; j< L ; j++)
			sommeSuiv += T[i][j];
			if (sommeSuiv != somme)
			OK = 0;
		i++; }
	// comparaison avec les colonnes
	j = 0;
	while (OK && j < C) {
		sommeSuiv = 0;
		for (i=0; i< C ; i++)
			sommeSuiv += T[i][j];
			if (sommeSuiv != somme)
			OK = 0;
		j++; }
	// comparaison avec les deux diagonales
	// la première
	if (OK) {
		sommeSuiv = 0;
		for (i=0; i< L ; i++)
		sommeSuiv += T[i][i];
		if (sommeSuiv != somme)
		OK = 0; }
	//la seconde
	if (OK) {
		sommeSuiv = 0;
		for (i=0; i< L ; i++)
		sommeSuiv += T[i][L -1 -i];
		if (sommeSuiv != somme)
		OK = 0; }
	return OK; 
	}
