char * ADN_Sequences (int N){
	char *T;
	int i;
	int k;
	char Tab[4]={'A','T','C','G' };
	for (i=0;i<N;i++){
		k = rand()%4;
		T[i]=Tab[k];
	}
	return T;
}

void Compte_Nucleotides(char* ch) {
	FILE *f;
	int nba=0, nbt=0, nbc=0, nbg=0;
	char c;
	f = fopen(ch, "r");
	if(f==NULL) {
		printf("\nErreur lors de l ouverture d un fichier");
		exit(1); 
	}
	while((c=fgetc(f))!=EOF){
		if(c=='C'){
			nbc++;
		}
		if(c=='A'){
			nba++;
		}if(c=='T'){
			nbt++;
		}if(c=='G'){
			nbg++;
		} 
	}
	printf("\n Le nombre des Cysteines est :%d\n",nbc);
	printf("\n Le nombre des Adenines est :%d\n",nba);
	printf("\n Le nombre des guanines est :%d\n",nbg);
	printf("\n Le nombre des Thymines est :%d\n",nbt);
	fclose(f);
}
