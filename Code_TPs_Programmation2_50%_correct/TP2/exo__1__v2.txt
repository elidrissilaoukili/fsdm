void affiche(char * ADN, int N){
	int i;
	printf("\n");
	for(i=0;i<N;i++) {
		printf("%c",*(ADN+i)); 
	}
	printf("\n\n");
}

void Compte_Nucleotides(char* ADN,int N){
	int i;
	int nbc=0,nba=0,nbt=0,nbg=0;
	for(i=0;i<N;i++){
		if(ADN[i]=='C'){
		nbc++;
		}
		if(ADN[i]=='A'){
		nba++;
		}
		if(ADN[i]=='T'){
		nbt++;
		}if(ADN[i]=='G'){
		nbg++;
		} 
	}
	printf("Le nombre des Cysteines est :%d\n",nbc);
	printf("Le nombre des Adenines est :%d\n",nba);
	printf("Le nombre des guanines est :%d\n",nbg);
	printf("Le nombre des Thymines est :%d\n",nbt);
}

char* Complement(char* ADN, int N){
	int i;
	for(i=0;i<N;i++){
		if(ADN[i]=='C'){
			ADN[i]='G';
			continue;
		}
		if(ADN[i]=='G'){
			ADN[i]='C';
			continue;
			}
		if(ADN[i]=='A'){
			ADN[i]='T';
			continue;
		}
		if(ADN[i]=='T'){
			ADN[i]='A';
			continue;
		} 
	}
	return ADN;
}

void codons(char* ADN, int * N){
	int i;

		if(i%3==0) printf("  ");

}
