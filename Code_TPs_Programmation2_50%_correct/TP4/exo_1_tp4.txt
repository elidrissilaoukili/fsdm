///////////////////////Question2///////////////////////////////
double DistancePoint(Point A, Point B){
	float d = sqrt(fabs((B.x-A.x)*(B.x-A.x)) + fabs((B.y-A.y)*(B.y-A.y))); 
	return d;
	}

typedef struct triangle{
	Point A;
	Point B;
	Point C;
	}Triangle;

Triangle tri;

Triangle getTriangle()
	{
	printf("Entrez les coordonnees du point A:\n");
	scanf("%f%f",&tri.A.x,&tri.A.y);
	printf("Entrez les coordonnees du point B:\n");
	scanf("%f%f",&tri.B.x,&tri.B.y);
	printf("Entrez les coordonnees du point C:\n");
	scanf("%f%f",&tri.C.x,&tri.C.y);
	}

//////////////////////Question6///////////////////////////////
float AireTriangle(Triangle tri){
	float partie_1 = ((tri.B.x)-(tri.A.x))*((tri.C.y)-(tri.A.y));
	float partie_2 = ((tri.C.x)-(tri.A.x))*((tri.B.y)-(tri.A.y));
	float aire = (fabs(partie_1-partie_2))/2;
	return aire; 
	}

float findArea(Rectangle r)
{
	return fabs((r.botRight.x - r.topLeft.x) * (r.topLeft.y - r.botRight.y));
}
